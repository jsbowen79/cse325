 Item 3: A text copy of the working Sales summary function: 
My sales summary function utilizes a total of 4 functions as shown below. 

(allSalesFiles, salesFiles) = FindFiles(storesDirectory);
createSalesSummaryFile();

void createSalesSummaryFile()
{
    (double totalSales, string detailString) = detailsStringWithTotal();
    string reportIntro = SalesSummary(totalSales);
    string completedReport = $"{reportIntro}\n{detailString}";
    Console.WriteLine($"completedReport: \n{completedReport}");
    File.WriteAllText(Path.Combine(salesTotalDir, "salesSummaryReport.txt"), completedReport);
}

string SalesSummary(double totalSales)
{
    List<string> report = new(
   [$"\t\t\t\t\tSales Summary{Environment.NewLine}", $"{new string('-', 96)}{Environment.NewLine}",
   $"{"Total Sales", -75 }     :{totalSales, 15:C}\n"]);
    return string.Join("", report);
}

(double, string) detailsStringWithTotal()
{
    List<string> report = new();
    report.Add($"Details: {Environment.NewLine}");
    double totalSales = 0;

    foreach (var file in salesFiles)
    {
        string salesJson = File.ReadAllText(file);
        SalesData? data = JsonConvert.DeserializeObject<SalesData?>(salesJson);
        totalSales += data?.Total ?? 0;
        report.Add($"{Path.GetRelativePath(storesDirectory, file),-80}:{data?.Total ?? 0,15:C}{Environment.NewLine}");
    }
    return (totalSales, string.Join("", report));
}

(List<string>, List<string>) FindFiles(string folderName)
{
    var foundFiles = Directory.EnumerateFiles(folderName, "*", SearchOption.AllDirectories);
    foreach (var file in foundFiles)
    {
        var extension = Path.GetExtension(file);
        if (extension == ".json")
        {
            allSalesFiles.Add(file);
        }
        if (file.EndsWith("sales.json"))
        {
            salesFiles.Add(file);
        }
    }

    return (allSalesFiles, salesFiles);
}
